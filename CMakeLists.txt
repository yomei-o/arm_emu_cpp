# Parity with build.sh, for MSVC and IDE builds.
cmake_minimum_required(VERSION 3.16)
project(arm_emu_cpp CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
file(GLOB SRC ${CMAKE_SOURCE_DIR}/src/*.cpp ${CMAKE_SOURCE_DIR}/src/jit/*.cpp)
add_executable(armemu ${SRC})
target_include_directories(armemu PRIVATE src)
if(MSVC)
  target_compile_options(armemu PRIVATE /utf-8 /EHsc /W3)
  target_compile_definitions(armemu PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(armemu PRIVATE -Wall -Wextra)
endif()
